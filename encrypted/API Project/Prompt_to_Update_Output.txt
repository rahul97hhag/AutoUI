"""Step-scoped network capture for Playwright pages.

NetworkRecorder.attach() registers request/response listeners once per page
(see before_scenario in features/environment.py). Behave's before_step
clears the recorded list and after_step reads it back, so each step's
Allure attachment only reflects network traffic that happened during that
specific step.

Playwright's sync API dispatches these events off the main thread, so the
listeners here only do cheap, synchronous bookkeeping -- storing headers and
object references that are already available without blocking. Anything
that can block (reading a response body) is deferred to resolve_body(),
which callers invoke from the main thread once the step has moved on.
"""
import threading

KEEP_RESOURCE_TYPES = {"xhr", "fetch", "document"}


class NetworkRecorder:
    def __init__(self):
        self._lock = threading.Lock()
        self._entries = []

    def attach(self, page):
        page.on("request", self._on_request)
        page.on("response", self._on_response)

    def _on_request(self, request):
        if request.resource_type not in KEEP_RESOURCE_TYPES:
            return
        entry = {
            "method": request.method,
            "url": request.url,
            "resource_type": request.resource_type,
            "request_headers": dict(request.headers),
            "request_post_data": request.post_data,
            "response": None,
        }
        with self._lock:
            self._entries.append(entry)

    def _on_response(self, response):
        with self._lock:
            for entry in reversed(self._entries):
                if entry["response"] is None and entry["url"] == response.url:
                    entry["response"] = {
                        "status": response.status,
                        "status_text": response.status_text,
                        "response_headers": dict(response.headers),
                        "_response": response,
                    }
                    return

    def clear(self):
        with self._lock:
            self._entries = []

    def snapshot(self):
        """A shallow copy, safe to iterate on the main thread after the
        step has finished (new events for the *next* step may still be
        arriving concurrently, but they land in a fresh list after clear())."""
        with self._lock:
            return list(self._entries)


def resolve_body(response):
    """Safely read a response body. Must be called from the main thread,
    after the request/response pair has had a chance to finish -- i.e. from
    after_step or from a step definition, never from inside the
    request/response event callbacks above.

    Returns (body_text, error): error is None on success, otherwise a
    description of why the body couldn't be read.
    """
    try:
        failure = response.finished()
        if failure:
            return None, failure
    except Exception as exc:  # noqa: BLE001 - surface any Playwright error as data, not a crash
        return None, str(exc)

    try:
        return response.text(), None
    except Exception as exc:  # noqa: BLE001
        return None, str(exc)

"""Allure attachment helpers used by features/environment.py's step hooks
and by step definitions that want to assert on captured API responses.
"""
import json
import logging

import allure
from jsonpath_ng import parse as jsonpath_parse

from utils.network_recorder import resolve_body

SENSITIVE_HEADERS = {"authorization", "cookie", "set-cookie", "x-api-key", "x-auth-token"}
REDACTED = "***"


def redact_headers(headers):
    return {key: (REDACTED if key.lower() in SENSITIVE_HEADERS else value) for key, value in headers.items()}


class ListLogHandler(logging.Handler):
    """Collects formatted log records in memory for the duration of one
    step, so that step's output can be attached to Allure without re-reading
    the shared log file."""

    def __init__(self):
        super().__init__()
        self.records = []
        self.setFormatter(logging.Formatter("%(asctime)s [%(levelname)s] %(name)s: %(message)s"))

    def emit(self, record):
        self.records.append(self.format(record))

    def as_text(self):
        return "\n".join(self.records) if self.records else "(no log output for this step)"


def attach_screenshot(page, name="screenshot"):
    try:
        png_bytes = page.screenshot(full_page=True)
    except Exception as exc:  # noqa: BLE001 - never let attachment capture fail the step itself
        allure.attach(str(exc), name=f"{name}-error", attachment_type=allure.attachment_type.TEXT)
        return
    allure.attach(png_bytes, name=name, attachment_type=allure.attachment_type.PNG)


def attach_logs(handler, name="step_logs"):
    allure.attach(handler.as_text(), name=name, attachment_type=allure.attachment_type.TEXT)


def _try_parse_json(raw):
    if not raw:
        return raw
    try:
        return json.loads(raw)
    except (TypeError, ValueError):
        return raw


def serialize_network_entries(entries):
    """Resolve response bodies and build a JSON-safe list for the full
    per-step network trace. Reads each body at most once per entry."""
    serialized = []
    for entry in entries:
        response_meta = entry["response"]
        data = {
            "method": entry["method"],
            "url": entry["url"],
            "resource_type": entry["resource_type"],
            "request_headers": redact_headers(entry["request_headers"]),
            "request_post_data": _try_parse_json(entry["request_post_data"]),
        }
        if response_meta is None:
            data["response"] = None
        else:
            body_text, body_error = resolve_body(response_meta["_response"])
            data["response"] = {
                "status": response_meta["status"],
                "status_text": response_meta["status_text"],
                "response_headers": redact_headers(response_meta["response_headers"]),
                "body": _try_parse_json(body_text),
                "body_error": body_error,
            }
        serialized.append(data)
    return serialized


def attach_network_trace(entries, name="network_trace"):
    serialized = serialize_network_entries(entries)
    allure.attach(json.dumps(serialized, indent=2, default=str), name=name, attachment_type=allure.attachment_type.JSON)
    return serialized


def _find_main_response(entries, which):
    candidates = [e for e in entries if e["resource_type"] in ("xhr", "fetch") and e["response"] is not None]
    if which == "last_xhr":
        return candidates[-1] if candidates else None
    return next((e for e in reversed(candidates) if which in e["url"]), None)


def _attach_verification_table(rows):
    columns = ["endpoint", "exp_status", "act_status", "jsonpath", "expected", "actual", "result"]
    widths = [50, 10, 10, 22, 18, 18, 6]
    lines = ["  ".join(c.ljust(w) for c, w in zip(columns, widths))]
    lines.append("-" * sum(widths))
    for row in rows:
        lines.append("  ".join(str(v).ljust(w) for v, w in zip(row, widths)))
    if not rows:
        lines.append("(no checks were requested -- pass expected_status and/or jsonpath_checks)")
    allure.attach("\n".join(lines), name="verification_result", attachment_type=allure.attachment_type.TEXT)


def assert_response(context, expected_status=None, jsonpath_checks=None, which="last_xhr"):
    """Verify the "main" API response captured so far during the current
    step, and attach a PASS/FAIL table to Allure either way.

    context: the Behave step context (must have .network_recorder and .page,
             both set up in before_scenario/before_step).
    which:   "last_xhr" (default) picks the most recently captured XHR/fetch
             response; any other string is matched as a substring of the URL.
    """
    entries = context.network_recorder.snapshot()
    entry = _find_main_response(entries, which)

    if entry is None:
        _attach_verification_table([])
        attach_screenshot(context.page, name="assertion_failure")
        raise AssertionError("assert_response: no matching XHR/fetch response was captured during this step")

    endpoint = entry["url"]
    response_meta = entry["response"]
    body_text, body_error = resolve_body(response_meta["_response"])
    body_json = None
    if body_text is not None:
        try:
            body_json = json.loads(body_text)
        except ValueError:
            body_json = None

    rows = []
    failures = []

    if expected_status is not None:
        actual_status = response_meta["status"]
        passed = actual_status == expected_status
        rows.append((endpoint, expected_status, actual_status, "-", "-", "-", "PASS" if passed else "FAIL"))
        if not passed:
            failures.append(f"{endpoint}: expected status {expected_status}, got {actual_status}")

    for path, expected_value in (jsonpath_checks or {}).items():
        matches = jsonpath_parse(path).find(body_json) if body_json is not None else []
        actual_value = matches[0].value if matches else None
        passed = bool(matches) and actual_value == expected_value
        rows.append((endpoint, "-", "-", path, repr(expected_value), repr(actual_value), "PASS" if passed else "FAIL"))
        if not passed:
            failures.append(f"{endpoint}: {path} expected {expected_value!r}, got {actual_value!r}")

    if body_error:
        failures.append(f"{endpoint}: response body could not be read ({body_error})")

    _attach_verification_table(rows)

    if failures:
        attach_screenshot(context.page, name="assertion_failure")
        raise AssertionError("; ".join(failures))

    return {"url": endpoint, "status": response_meta["status"], "body": body_json}

"""Step definitions demonstrating assert_response() (utils/allure_helpers.py)
against a response the running page genuinely receives.

automationexercise.com's own UI is server-rendered -- its pages don't fire
XHR/fetch calls on their own -- but the site also publishes a small REST API
for practice at /api_list (e.g. GET /api/productsList ->
{"responseCode": 200, "products": [...]}). Calling it via page.evaluate()'s
fetch() makes it a real request/response the current page's
NetworkRecorder observes the same way it would observe any API call the
site's own JS made, so this exercises the full Phase 1-3 pipeline
end-to-end rather than mocking it.
"""
import json

from behave import when, then

from utils.allure_helpers import assert_response


@when('I call the GET API "{path}" from the page')
def step_call_api(context, path):
    url = path if path.startswith("http") else f"{context.config_data['base_url']}{path}"
    context.page.evaluate(f"() => fetch('{url}')")


@then("the API response should have status {expected_status:d}")
def step_check_status_only(context, expected_status):
    assert_response(context, expected_status=expected_status)


@then('the API response field "{json_path}" should equal {expected_value}')
def step_check_jsonpath(context, json_path, expected_value):
    try:
        expected = json.loads(expected_value)
    except ValueError:
        expected = expected_value
    assert_response(context, jsonpath_checks={json_path: expected})

Found it — a real architectural bug, not a flaky test. before_step's network_recorder.clear() wipes the recorder before every single step, including the Then step. So the "When I call the API" step's captured response gets deleted the instant the next step (Then...) starts, before assert_response ever runs. The standalone script above proves the recorder itself works fine in isolation — this is purely a step-boundary design flaw in how I wired Phase 2's "clear before each step" into a workflow where the call and the assertion are in separate Gherkin steps.

Real fix: stop destroying history on every step boundary. Instead, track a per-step mark (an index into an append-only list) — after_step's per-step JSON attachment uses since(mark) (still correctly scoped to just that step), while assert_response can see the whole scenario's history, so it can find a response from an earlier step.
